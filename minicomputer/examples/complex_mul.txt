% Lab 1 example program: (a + bi) * (c + di) = (ac - bd) + (ad + bc)i
% data: a at 0, b at 4, c at 8, d at 12; result: real part at 16, imaginary part at 20
x1 = [0]
x2 = [4]
x3 = [8]
x4 = [12]
x5 = x1 * x3      % ac
x6 = x2 * x4      % bd
x7 = x5 - x6      % real
x8 = x1 * x4      % ad
x9 = x2 * x3      % bc
x10 = x8 + x9     % imaginary
[16] = x7
[20] = x10
