% count_key.txt - count the occurrences of a key in an array
% Task:   Write a program that counts how many times a key occurs in an array.
% Data:   N at 0; key at 4; A[0..N-1] at 8 .. 4N + 4
% Result: the count at 4N + 8
%         count_key_data.byte: N = 12, key 5 -> [56] = 4 (the -5 in the data does not count)
% Method: counted loop; equality test x9 = A[i] - key, then BNE skips the increment.
% Regs:   x1 N, x2 i, x3 &A[i], x4 A[i], x5 key, x10 count
x1 = [0]          % N
x5 = [4]          % key
x2 = 0            % i
x3 = 8            % &A[0]
x10 = 0           % count
.loop
x9 = x2 - x1
BGE .done         % i >= N
x4 = [x3]
x9 = x4 - x5
BNE .next         % A[i] != key
x10 = x10 + 1
.next
x3 = x3 + 4
x2 = x2 + 1
BAL .loop
.done
[x3] = x10        % x3 = 4N + 8, just past the array
