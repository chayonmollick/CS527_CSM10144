% average.txt - integer average of an array (sum, average, remainder)
% Task:   Write a program that computes the sum of N numbers, their integer average
%         and the remainder of the division.
% Data:   N at 0 (N >= 1); A[0..N-1] at 4 .. 4N
% Result: sum at 4N + 4, average = sum / N at 4N + 8, remainder at 4N + 12
%         average_data.byte : N = 7, sum 123 -> [32] = 123, [36] = 17, [40] = 4
%         average_data2.byte: N = 4, sum -15 -> [20] = -15, [24] = -3, [28] = -3
% Method: counted loop for the sum, then one division. Division truncates towards
%         zero: -15 / 4 = -3 (not -4), and the remainder has the sign of the sum.
%         For a rounded average of a positive sum use (sum + N / 2) / N.
% Regs:   x1 N, x2 i, x3 &A[i], x4 A[i], x10 sum, x11 average, x12 remainder
x1 = [0]          % N
x2 = 0            % i
x3 = 4            % &A[i]
x10 = 0           % sum
.loop
x9 = x2 - x1
BGE .done         % i >= N
x4 = [x3]
x10 = x10 + x4    % sum = sum + A[i]
x3 = x3 + 4
x2 = x2 + 1
BAL .loop
.done
x11 = x10 / x1    % average (truncated towards zero)
x12 = x11 * x1
x12 = x10 - x12   % remainder = sum - average * N
[x3] = x10        % x3 = 4N + 4: sum
x3 = x3 + 4
[x3] = x11        % 4N + 8: average
x3 = x3 + 4
[x3] = x12        % 4N + 12: remainder
