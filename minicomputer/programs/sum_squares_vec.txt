% sum_squares_vec.txt - sum of squares of an array, with vector instructions
% Task:   Write a program that computes A[0]^2 + A[1]^2 + ... + A[N-1]^2 using
%         vector instructions. N is any number >= 0 (not only a multiple of 8).
% Data:   N at 0; A[0..N-1] at 4 .. 4N
% Result: the sum at 4N + 4. The 8 words after it (4N + 8 ..) are scratch space
%         for adding the lanes.
%         sum_squares_vec_data.byte : N = 13 (one block of 8 + 5 left over) -> [56] = 377
%         sum_squares_vec_data2.byte: N = 16 = 1..16 (two blocks, nothing left) -> [68] = 1496
% Method: VECTOR. Each block of 8 is squared with one vector multiply and added
%         into 8 partial sums (the lanes of v0). The N mod 8 leftover elements go
%         through a scalar loop. The 8 lanes are then added with a vector
%         "shift and add": store v0, load it back 4 words further on and add
%         (lanes 0..3 now hold pair sums), then 2 words, then 1 word: lane 0 is the
%         total. That is 3 vector adds instead of 7 scalar ones.
% Variants: plain vector sum of an array -> delete "v1 = v1 * v1" and "x4 = x4 * x4".
% Regs:   x1 N, x2 i, x3 &A[i], x7 N - 8, x10 scalar sum, x20..x23 scratch addresses,
%         v0 8 partial sums, v1 data block
x1 = [0]          % N
x2 = 0            % i
x3 = 4            % &A[i]
x7 = x1 - 8       % a full block of 8 starts at i <= N - 8
v0 = v0 - v0      % partial sums = 0 (any vector minus itself)
.vloop
x9 = x2 - x7
BGT .tail         % i > N - 8: fewer than 8 elements left
v1 = [x3]         % A[i .. i+7]
v1 = v1 * v1      % 8 squares at once
v0 = v0 + v1      % add them to the 8 partial sums
x3 = x3 + 32      % next block: 8 words = 32 bytes
x2 = x2 + 8
BAL .vloop
.tail
x10 = 0           % sum of the leftover elements
.sloop
x9 = x2 - x1
BGE .reduce       % i >= N
x4 = [x3]
x4 = x4 * x4
x10 = x10 + x4
x3 = x3 + 4
x2 = x2 + 1
BAL .sloop
.reduce
x20 = x3 + 4      % x3 = 4N + 4 (result); scratch S = 4N + 8
x21 = x20 + 16    % S + 16 (4 words on)
x22 = x20 + 8     % S + 8  (2 words on)
x23 = x20 + 4     % S + 4  (1 word on)
[x20] = v0        % lanes l0 .. l7 at S
v1 = [x21]        % l4 l5 l6 l7 and 4 words we do not care about
v0 = v0 + v1      % lanes 0..3: l0+l4, l1+l5, l2+l6, l3+l7
[x20] = v0
v1 = [x22]
v0 = v0 + v1      % lanes 0..1: l0+l4+l2+l6, l1+l5+l3+l7
[x20] = v0
v1 = [x23]
v0 = v0 + v1      % lane 0: the sum of all 8 lanes
[x20] = v0
x11 = [x20]       % read lane 0
x10 = x10 + x11   % + the leftover elements
[x3] = x10        % result at 4N + 4
