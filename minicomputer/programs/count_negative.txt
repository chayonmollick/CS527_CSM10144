% count_negative.txt - count the negative, zero and positive elements of an array
% Task:   Write a program that counts how many elements of an array are negative,
%         how many are zero and how many are positive.
% Data:   N at 0; A[0..N-1] at 4 .. 4N
% Result: negatives at 4N + 4, zeros at 4N + 8, positives at 4N + 12
%         count_negative_data.byte: N = 10 -> [44] = 4, [48] = 2, [52] = 4
% Method: a load does not set the flags, so "x9 = a - 0" copies a's sign and zero
%         test into the flags; then BMI (negative) and BEQ (zero) read the same
%         flags one after the other (a branch does not change them).
% Variants: count elements > K -> x9 = a - K, then BGT (K in a register if > 255).
% Regs:   x1 N, x2 i, x3 &A[i], x4 A[i], x10 negatives, x11 zeros, x12 positives
x1 = [0]          % N
x2 = 0            % i
x3 = 4            % &A[i]
x10 = 0           % negatives
x11 = 0           % zeros
x12 = 0           % positives
.loop
x9 = x2 - x1
BGE .done         % i >= N
x4 = [x3]         % a = A[i]
x9 = x4 - 0       % set the flags from a
BMI .neg          % a < 0
BEQ .zero         % a == 0
x12 = x12 + 1     % a > 0
BAL .next
.neg
x10 = x10 + 1
BAL .next
.zero
x11 = x11 + 1
.next
x3 = x3 + 4
x2 = x2 + 1
BAL .loop
.done
[x3] = x10        % x3 = 4N + 4: negatives
x3 = x3 + 4
[x3] = x11        % 4N + 8: zeros
x3 = x3 + 4
[x3] = x12        % 4N + 12: positives
