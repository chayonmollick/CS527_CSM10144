% count_even.txt - count the even and the odd elements of an array
% Task:   Write a program that counts how many elements of an array are even and
%         how many are odd.
% Data:   N at 0; A[0..N-1] at 4 .. 4N
% Result: number of even elements at 4N + 4, number of odd elements at 4N + 8
%         count_even_data.byte: N = 10 -> [44] = 6, [48] = 4
% Method: there is no modulus or AND, so r = a - (a / 2) * 2. Division truncates
%         towards zero, so r is 0 (even), 1 or -1 (odd negative numbers give -1):
%         test r == 0, never r == 1. The subtraction that makes r sets the Z flag.
% Variants: divisible by k -> divide and multiply by k instead of 2.
% Regs:   x1 N, x2 i, x3 &A[i], x4 A[i], x5 remainder, x10 even count, x11 odd count
x1 = [0]          % N
x2 = 0            % i
x3 = 4            % &A[i]
x10 = 0           % even count
.loop
x9 = x2 - x1
BGE .done         % i >= N
x4 = [x3]         % a = A[i]
x5 = x4 / 2
x5 = x5 * 2
x5 = x4 - x5      % r = a - (a / 2) * 2; sets Z when r == 0
BNE .next         % r != 0: odd
x10 = x10 + 1     % even
.next
x3 = x3 + 4
x2 = x2 + 1
BAL .loop
.done
[x3] = x10        % x3 = 4N + 4: even count
x11 = x1 - x10    % odd = N - even
x3 = x3 + 4
[x3] = x11        % odd count at 4N + 8
