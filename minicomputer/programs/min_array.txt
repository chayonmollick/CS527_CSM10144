% min_array.txt - smallest element of an array and its index (pointer loop)
% Task:   Write a program that finds the smallest element of an array of N signed
%         integers and the index of its first occurrence.
% Data:   N at 0 (N >= 1); A[0..N-1] at 4 .. 4N
% Result: min at 4N + 4, its index at 4N + 8
%         min_array_data.byte: N = 8 -> [36] = -7, [40] = 2 (-7 is at index 2 and 4)
% Method: instead of counting i, the loop walks a pointer from &A[1] up to the end
%         address 4N + 4; the index is recovered from the address: (address - 4) / 4.
% Regs:   x1 N, x3 &A[i], x4 A[i], x5 end address, x9 compare, x10 min, x11 index of min
x1 = [0]          % N
x5 = x1 * 4
x5 = x5 + 4       % end = 4N + 4, the first address after the array
x3 = 4            % &A[0]
x10 = [x3]        % min = A[0]
x11 = 0           % its index
.loop
x3 = x3 + 4       % next element
x9 = x3 - x5
BGE .done         % the pointer reached the end
x4 = [x3]
x9 = x4 - x10     % A[i] - min
BGE .loop         % A[i] >= min: keep the old min
x10 = x4          % new min
x11 = x3 - 4      % index = (address - 4) / 4
x11 = x11 / 4
BAL .loop
.done
[x5] = x10        % min at 4N + 4
x5 = x5 + 4
[x5] = x11        % index at 4N + 8
