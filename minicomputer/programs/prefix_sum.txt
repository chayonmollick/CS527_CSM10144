% prefix_sum.txt - prefix sums (running totals) of an array
% Task:   Write a program that computes the prefix sums S[i] = A[0] + A[1] + ... + A[i]
%         of an array A of N elements into a second array S.
% Data:   N at 0; A[0..N-1] at 4 .. 4N; S is written right after A, at 4 + 4N
% Result: S[0..N-1] at 4 + 4N ..
%         prefix_sum_data.byte: N = 10, A = 3 1 4 1 5 9 2 6 5 3
%                               -> S = 3 4 8 9 14 23 25 31 36 39 at 44 .. 80
% Method: counted loop with a running sum: S[i] = S[i-1] + A[i], one add per
%         element. It is sequential (each S[i] needs the one before), so scalar.
% Variants: in place -> store to x3 instead of x4 (A[i] = running sum).
% Regs:   x1 N, x2 i, x3 &A[i], x4 &S[i], x5 A[i], x10 running sum
x1 = [0]          % N
x2 = 0            % i
x3 = 4            % &A[0]
x4 = x1 * 4
x4 = x4 + 4       % &S[0] = 4 + 4N
x10 = 0           % running sum
.loop
x9 = x2 - x1
BGE .done         % i >= N
x5 = [x3]         % A[i]
x10 = x10 + x5    % sum of A[0..i]
[x4] = x10        % S[i]
x3 = x3 + 4
x4 = x4 + 4
x2 = x2 + 1
BAL .loop
.done
