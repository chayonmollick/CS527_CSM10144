% scale_offset_vec.txt - y = a*x + b for every element, with vector instructions
% Task:   Write a program that computes Y[i] = a * X[i] + b for an array X of N
%         elements using vector instructions. N need not be a multiple of 8.
% Data:   N at 0; a at 4; b at 8; X[0..N-1] at 12 .. 4N + 8; Y is written right
%         after X, at 12 + 4N
% Result: Y[0..N-1] at 12 + 4N ..
%         scale_offset_vec_data.byte : N = 16, a = 3, b = 5     -> Y at 76 .. 136
%         scale_offset_vec_data2.byte: N = 13, a = -2, b = 7, X = 1..13 -> Y = 5 3 1 .. -19 at 64 .. 112
% Method: VECTOR. a and b stay in scalar registers: "v1 = v1 * x5" multiplies all
%         8 lanes by a and "v1 = v1 + x6" adds b to all 8 lanes (vector op scalar
%         register). Blocks of 8 while at least 8 are left, then a scalar tail.
% Variants: C = A + B (array add) for any N -> load a second block and add vectors.
% Regs:   x1 N, x2 &X[i], x3 &Y[i], x4 i, x5 a, x6 b, x7 N - 8, x8 one element, v1 a block
x1 = [0]          % N
x5 = [4]          % a
x6 = [8]          % b
x2 = 12           % &X[0]
x3 = x1 * 4
x3 = x3 + 12      % &Y[0] = 12 + 4N
x4 = 0            % i
x7 = x1 - 8       % last i at which a full block of 8 still fits
.vloop
x9 = x4 - x7
BGT .tail         % fewer than 8 left
v1 = [x2]         % X[i .. i+7]
v1 = v1 * x5      % * a  (every lane)
v1 = v1 + x6      % + b  (every lane)
[x3] = v1         % Y[i .. i+7]
x2 = x2 + 32
x3 = x3 + 32
x4 = x4 + 8
BAL .vloop
.tail
x9 = x4 - x1
BGE .done         % i >= N
x8 = [x2]         % one element at a time
x8 = x8 * x5
x8 = x8 + x6
[x3] = x8
x2 = x2 + 4
x3 = x3 + 4
x4 = x4 + 1
BAL .tail
.done
