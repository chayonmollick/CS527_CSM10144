% abs_array.txt - absolute value of every element
% Task:   Write a program that replaces every element of an array by its absolute
%         value.
% Data:   N at 0; A[0..N-1] at 4 .. 4N
% Result: |A[i]| in the same place, 4 .. 4N
%         abs_array_data.byte: N = 10, 3 -7 0 -1 15 -128 42 -9 6 -2 -> 3 7 0 1 15 128 42 9 6 2
% Method: counted loop. Sign test: "x9 = a - 0" then BPL (a load does not set the
%         flags). Negate with 0 - a, using a register that holds 0 (constants
%         cannot be the first operand). Scalar on purpose: each element needs its
%         own decision and there is no vector compare or select.
% Regs:   x1 N, x2 i, x3 &A[i], x4 A[i], x20 zero
x1 = [0]          % N
x2 = 0            % i
x3 = 4            % &A[i]
x20 = 0           % 0, for 0 - a
.loop
x9 = x2 - x1
BGE .done         % i >= N
x4 = [x3]         % a = A[i]
x9 = x4 - 0       % flags from a
BPL .next         % a >= 0: already its absolute value
x4 = x20 - x4     % a = 0 - a
[x3] = x4         % write it back
.next
x3 = x3 + 4
x2 = x2 + 1
BAL .loop
.done
