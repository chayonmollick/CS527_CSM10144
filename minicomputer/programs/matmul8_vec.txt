% matmul8_vec.txt - C = A x B for 8x8 matrices, with vector instructions
% Task:   Write a program that multiplies two 8x8 matrices using vector instructions.
% Data:   A (row-major, 64 words) at 0 .. 252; B at 256 .. 508
%         (the data file holds A[i][j] = (ij + 2i + 3j) mod 9 - 4 and
%          B[i][j] = (3i + 7j) mod 10 - 4, small numbers from -4 to 5)
% Result: C (row-major) at 512 .. 764, row i at 512 + 32i
%         matmul8_vec_data.byte -> row 0 = -8 -35 8 21 -36 -23 50 -37,
%                                  row 7 = 20 -20 -30 -20 20 0 0 30
% Method: VECTOR. Row i of C is a combination of the rows of B:
%             C[i][*] = A[i][0] * B[0][*] + A[i][1] * B[1][*] + ... + A[i][7] * B[7][*]
%         The 8 rows of B are loaded once into v1..v8. Per row of C: 8 scalar loads
%         of A[i][k], 8 "vector * scalar register" multiplies, 7 vector adds and
%         one vector store. No lane sums are needed, unlike matvec_vec.txt.
% Regs:   x1 &A[i][k] (walks through A in order), x3 &C[i], x4 i, x5 A[i][k],
%         v1..v8 rows of B, v0 row of C being built, v9 one term
x2 = 128
x2 = x2 * 2       % &B = 256
v1 = [x2]         % B row 0
x2 = x2 + 32
v2 = [x2]         % B row 1
x2 = x2 + 32
v3 = [x2]         % B row 2
x2 = x2 + 32
v4 = [x2]         % B row 3
x2 = x2 + 32
v5 = [x2]         % B row 4
x2 = x2 + 32
v6 = [x2]         % B row 5
x2 = x2 + 32
v7 = [x2]         % B row 6
x2 = x2 + 32
v8 = [x2]         % B row 7
x1 = 0            % &A[0][0]
x3 = 128
x3 = x3 * 4       % &C[0] = 512
x4 = 0            % row i
.row
x9 = x4 - 8
BGE .done         % 8 rows done
x5 = [x1]         % A[i][0]
v0 = v1 * x5      % C row = A[i][0] * B row 0
x1 = x1 + 4
x5 = [x1]         % A[i][1]
v9 = v2 * x5
v0 = v0 + v9      % + A[i][1] * B row 1
x1 = x1 + 4
x5 = [x1]         % A[i][2]
v9 = v3 * x5
v0 = v0 + v9
x1 = x1 + 4
x5 = [x1]         % A[i][3]
v9 = v4 * x5
v0 = v0 + v9
x1 = x1 + 4
x5 = [x1]         % A[i][4]
v9 = v5 * x5
v0 = v0 + v9
x1 = x1 + 4
x5 = [x1]         % A[i][5]
v9 = v6 * x5
v0 = v0 + v9
x1 = x1 + 4
x5 = [x1]         % A[i][6]
v9 = v7 * x5
v0 = v0 + v9
x1 = x1 + 4
x5 = [x1]         % A[i][7]
v9 = v8 * x5
v0 = v0 + v9      % + A[i][7] * B row 7
x1 = x1 + 4       % &A[i+1][0]
[x3] = v0         % row i of C
x3 = x3 + 32
x4 = x4 + 1
BAL .row
.done
