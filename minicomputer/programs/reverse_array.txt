% reverse_array.txt - reverse an array in place
% Task:   Write a program that reverses an array of N elements in place
%         (A[0] <-> A[N-1], A[1] <-> A[N-2], ...).
% Data:   N at 0; A[0..N-1] at 4 .. 4N
% Result: the reversed array in the same place, 4 .. 4N
%         reverse_array_data.byte : N = 9 (odd: the middle element stays) 10 20 .. 90 -> 90 80 .. 10
%         reverse_array_data2.byte: N = 8 (even) 1 .. 8 -> 8 .. 1
% Method: while loop with two pointers, lo from the front and hi from the back:
%         swap A[lo] and A[hi] while lo < hi, then lo moves up and hi moves down.
% Variants: reversed copy into B -> read A forwards, write B backwards (no swap).
% Regs:   x1 N, x3 lo address, x4 hi address, x5 A[lo], x6 A[hi]
x1 = [0]          % N
x3 = 4            % lo = &A[0]
x4 = x1 * 4       % hi = &A[N-1] = 4N
.loop
x9 = x3 - x4
BGE .done         % lo >= hi: the pointers met in the middle
x5 = [x3]         % swap A[lo] and A[hi]
x6 = [x4]
[x3] = x6
[x4] = x5
x3 = x3 + 4       % lo moves up
x4 = x4 - 4       % hi moves down
BAL .loop
.done
