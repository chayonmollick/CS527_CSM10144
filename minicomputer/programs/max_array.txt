% max_array.txt - largest element of an array and its index
% Task:   Write a program that finds the largest element of an array of N signed
%         integers and the index (0-based) of its first occurrence.
% Data:   N at address 0 (N >= 1); A[0..N-1] at 4, 8, ..., 4N
% Result: max at 4N + 4, its index at 4N + 8
%         max_array_data.byte : N = 10 -> [44] = 37, [48] = 2 (37 is at index 2 and 4)
%         max_array_data2.byte: N = 6, all negative -> [28] = -3, [32] = 1
% Method: scalar counted loop. Compare = subtract (x9 = A[i] - max), then a signed
%         branch (BLE). Not vectorisable: the machine has no vector compare.
% Variants: minimum -> BGE instead of BLE; last occurrence -> BLT instead of BLE.
% Regs:   x1 N, x2 i, x3 &A[i], x4 A[i], x9 compare, x10 max, x11 index of max
x1 = [0]          % N
x3 = 4            % x3 = &A[0]
x10 = [x3]        % max = A[0]
x11 = 0           % its index
x2 = 1            % i = 1 (A[0] is already taken)
.loop
x9 = x2 - x1
BGE .done         % i >= N: every element seen
x3 = x3 + 4       % &A[i]
x4 = [x3]         % A[i]
x9 = x4 - x10     % A[i] - max
BLE .next         % A[i] <= max: keep the old max (so the first occurrence wins)
x10 = x4          % new max
x11 = x2          % and its index
.next
x2 = x2 + 1
BAL .loop
.done
x3 = x3 + 4       % x3 was &A[N-1] = 4N, now 4N + 4
[x3] = x10        % max
x3 = x3 + 4
[x3] = x11        % index
