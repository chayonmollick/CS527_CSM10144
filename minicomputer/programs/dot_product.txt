% dot_product.txt - dot product of two arrays (scalar loop)
% Task:   Write a program that computes the dot product of two arrays A and B of N
%         elements: A[0]*B[0] + A[1]*B[1] + ... + A[N-1]*B[N-1].
% Data:   N at 0; A at 4 .. 4N; B right after A, at 4 + 4N .. 8N
% Result: the dot product right after B, at 4 + 8N
%         dot_product_data.byte: N = 10 -> [84] = -15
% Method: scalar counted loop with two pointers that move together. The vector
%         version (8 products per instruction) is dot_product_vec.txt.
% Regs:   x1 N, x2 i, x3 &A[i], x4 &B[i], x5 A[i] (then the product), x6 B[i], x10 sum
x1 = [0]          % N
x2 = 0            % i
x3 = 4            % &A[0]
x4 = x1 * 4
x4 = x4 + 4       % &B[0] = 4 + 4N
x10 = 0           % sum
.loop
x9 = x2 - x1
BGE .done         % i >= N
x5 = [x3]         % A[i]
x6 = [x4]         % B[i]
x5 = x5 * x6
x10 = x10 + x5    % sum += A[i] * B[i]
x3 = x3 + 4
x4 = x4 + 4
x2 = x2 + 1
BAL .loop
.done
[x4] = x10        % x4 = 4 + 8N, just past B
