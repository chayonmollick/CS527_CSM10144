% histogram.txt - histogram of values 0..7
% Task:   Write a program that counts how many elements of an array are equal to
%         0, 1, 2, ..., 7 (a histogram with 8 bins). Values outside 0..7 are not
%         put in a bin but counted as invalid.
% Data:   N at 0; A[0..N-1] at 4 .. 4N
% Result: H[0..7] right after A, at 4N + 4 .. 4N + 32; the invalid count at 4N + 36
%         histogram_data.byte: N = 20 -> H = 2 3 4 2 1 4 1 1 at 84 .. 112, [116] = 2
% Method: the value picks the bin: &H[a] = H + 4a, then H[a] = H[a] + 1
%         (load, add, store through a computed address). Range check with two
%         branches (BMI for a < 0, BGT for a > 7). One vector store of a zero
%         vector clears all 8 bins at the start.
% Regs:   x1 N, x2 i, x3 &A[i], x4 a, x5 &H[a], x6 H[a], x11 invalid count, x20 &H[0], v0 zeros
x1 = [0]          % N
x20 = x1 * 4
x20 = x20 + 4     % &H[0] = 4N + 4
v0 = v0 - v0      % 8 zeros
[x20] = v0        % H[0..7] = 0 with one vector store
x11 = 0           % invalid count
x2 = 0            % i
x3 = 4            % &A[0]
.loop
x9 = x2 - x1
BGE .done         % i >= N
x4 = [x3]         % a = A[i]
x9 = x4 - 0
BMI .invalid      % a < 0
x9 = x4 - 7
BGT .invalid      % a > 7
x5 = x4 * 4
x5 = x5 + x20     % &H[a]
x6 = [x5]
x6 = x6 + 1
[x5] = x6         % H[a] = H[a] + 1
BAL .next
.invalid
x11 = x11 + 1
.next
x3 = x3 + 4
x2 = x2 + 1
BAL .loop
.done
x5 = x20 + 32     % right after the 8 bins
[x5] = x11        % invalid count
