% matvec_vec.txt - matrix-vector product y = M x (8x8), with vector instructions
% Task:   Write a program that multiplies an 8x8 matrix M by a vector x of 8
%         elements using vector instructions: y[i] = M[i][0]*x[0] + ... + M[i][7]*x[7].
% Data:   M (row-major, 64 words) at 0 .. 252; x at 256 .. 284
%         (the data file holds M[i][j] = i - j and x[j] = j + 1)
% Result: y at 288 .. 316; 320 .. 348 is scratch space for the lane sums
%         matvec_vec_data.byte -> y = -168 -132 -96 -60 -24 12 48 84 (y[i] = 36i - 168)
% Method: VECTOR. x is loaded once into v1. Per row: one vector load and one vector
%         multiply give the 8 products M[i][j] * x[j]; they are stored and their 8
%         lanes added with scalar loads (a row-major M gives no vector way to add
%         across lanes). The 8 lane addresses are computed once, before the loop.
%         If M were stored by columns (or the task were x^T M), y would instead be
%         a sum of columns times x[j] with no lane sums at all: see matmul8_vec.txt.
% Regs:   x1 &M[i][0], x3 &y[i], x4 i, x20..x27 scratch lane addresses, x10 y[i],
%         x11 one lane, v1 x, v2 products
x1 = 0            % &M[0][0]
x2 = 128
x2 = x2 * 2       % &x = 256
v1 = [x2]         % x[0..7], loaded once
x3 = x2 + 32      % &y[0] = 288
x20 = x3 + 32     % scratch = 320: lane k at 320 + 4k
x21 = x20 + 4
x22 = x20 + 8
x23 = x20 + 12
x24 = x20 + 16
x25 = x20 + 20
x26 = x20 + 24
x27 = x20 + 28
x4 = 0            % row i
.row
x9 = x4 - 8
BGE .done         % 8 rows done
v2 = [x1]         % row i of M
v2 = v2 * v1      % M[i][j] * x[j] for j = 0..7
[x20] = v2        % store the 8 products
x10 = [x20]       % add the 8 lanes
x11 = [x21]
x10 = x10 + x11
x11 = [x22]
x10 = x10 + x11
x11 = [x23]
x10 = x10 + x11
x11 = [x24]
x10 = x10 + x11
x11 = [x25]
x10 = x10 + x11
x11 = [x26]
x10 = x10 + x11
x11 = [x27]
x10 = x10 + x11
[x3] = x10        % y[i]
x1 = x1 + 32      % next row of M
x3 = x3 + 4       % next y
x4 = x4 + 1
BAL .row
.done
