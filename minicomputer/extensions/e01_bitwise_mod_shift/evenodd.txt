% e01: count the even and the odd numbers of an array, testing bit 0 with "& 1"
% data: N at 0, the numbers at 4 .. 4N; writes the even count at 4N+4 and the odd count at 4N+8
% (& 1 also works for negative numbers: -5 & 1 = 1; "mod 2" would give -1 for -5)
x1 = [0]          % numbers left
x2 = 4            % address of a[i]
x3 = 0            % evens
x4 = 0            % odds
.loop
x9 = x1 - 0
BEQ .done
x5 = [x2]
x6 = x5 & 1       % 0 = even, 1 = odd
x9 = x6 - 0       % & does not set the flags: compare with 0
BEQ .even
x4 = x4 + 1
BAL .next
.even
x3 = x3 + 1
.next
x2 = x2 + 4
x1 = x1 - 1
BAL .loop
.done
[x2] = x3         % even count at 4N+4
x2 = x2 + 4
[x2] = x4         % odd count at 4N+8
