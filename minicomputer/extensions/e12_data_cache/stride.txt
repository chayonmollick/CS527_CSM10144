% EXT e12 example: add up N words that are S bytes apart (the array starts at 16).
% data: N at 0, S (stride in bytes, a multiple of 4) at 4; the sum is written at 8.
% A cache line holds 16 bytes = 4 words:
%   S = 4:  one miss brings in the next 3 words too -> 3 of every 4 loads hit
%   S = 16: every load is in a new line             -> every load misses
x1 = [0]          % N
x2 = [4]          % S
x3 = 16           % address of the next word
x4 = 0            % i
x5 = 0            % sum
.loop
x6 = x4 - x1
BGE .done         % i >= N
x7 = [x3]
x5 = x5 + x7
x3 = x3 + x2
x4 = x4 + 1
BAL .loop
.done
[8] = x5
