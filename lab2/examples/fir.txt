% Lab 2 example program: FIR filter, the weights are constants in the program
%   y[n] = 1*x[n] + 2*x[n-1] + 3*x[n-2] + 4*x[n-3]      (inputs before x[0] count as 0)
% data: N at 0, the input x[0..N-1] at 4 .. 4N; the output y[0..N-1] is written from 0x100.
% The handout puts the output at 0x10, inside the input; 0x100 = 256 is clear of it for N <= 63.
x1 = [0]          % N
x2 = 4            % address of x[n]
x3 = 128
x3 = x3 * 2       % address of y[n] = 0x100 (a constant stops at 255)
x4 = 0            % n
x10 = 0           % x[n-1]   the filter's delay line
x11 = 0           % x[n-2]
x12 = 0           % x[n-3]
.loop
x5 = x4 - x1
BGE .done         % n >= N
x9 = [x2]         % x[n]
x6 = x9 * 1       % 1*x[n]
x7 = x10 * 2      % 2*x[n-1]
x6 = x6 + x7
x7 = x11 * 3      % 3*x[n-2]
x6 = x6 + x7
x7 = x12 * 4      % 4*x[n-3]
x6 = x6 + x7
[x3] = x6         % y[n]
x12 = x11         % shift the delay line
x11 = x10
x10 = x9
x2 = x2 + 4
x3 = x3 + 4
x4 = x4 + 1
BAL .loop
.done
