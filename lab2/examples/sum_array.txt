% Lab 2 handout example: sum of an array (the program below is the handout's, unchanged).
% data: size N at 0, elements at 4 .. 4N. The handout says the sum goes to N*4 + 4, but this
% program stores it at [x3] = N*4, over the last element (see README, Decisions).
x1 = 0       % index
x2 = 0        % sum
x3 = 0        % array address
x4 =  [x3]    %  read the size of array
.loopback
x15 = x1 - x4   % compare
BEQ .exit
x3 = x3 + 4    % increase the address
x6 =  [x3]       % load the array data
x2 = x2 + x6  % update sum
x1 = x1 + 1    % increment index
BAL .loopback
.exit
[x3] = x2
