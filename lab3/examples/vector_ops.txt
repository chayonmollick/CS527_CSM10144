% Lab 3: every vector instruction once (the handout's examples plus the other operand forms).
% data: 1 2 3 4 5 6 7 8 at 0..28. Each result (8 words) is written from 32 on; test.sh checks them.
v1 = [0]          % 2C  vector read, constant address: 1 2 3 4 5 6 7 8
x1 = 0
v2 = [x1]         % 25  vector read, address in x1:    1 2 3 4 5 6 7 8
v3 = v1 + v2      % 21  2 4 6 8 10 12 14 16
v4 = v3 - v1      % 22  1 2 3 4 5 6 7 8
v5 = v1 * v2      % 23  1 4 9 16 25 36 49 64
v6 = v1 + 15      % 29  16 17 18 19 20 21 22 23
v7 = v1 - 10      % 2A  -9 -8 -7 -6 -5 -4 -3 -2
v8 = v1 * 3       % 2B  3 6 9 12 15 18 21 24
x4 = 5
v9 = v1 + x4      % 31  6 7 8 9 10 11 12 13
v10 = v1 - x4     % 32  -4 -3 -2 -1 0 1 2 3
v11 = v1 * x4     % 33  5 10 15 20 25 30 35 40
v12 = v11         % copy (compiled as v12 = v11 + 0)
[32] = v3         % 2E  vector write, constant address
[64] = v4
x2 = 96
[x2] = v5         % 26  vector write, address in x2
x2 = x2 + 32
[x2] = v6         % 128
x2 = x2 + 32
[x2] = v7         % 160
x2 = x2 + 32
[x2] = v8         % 192
x2 = x2 + 32
[x2] = v9         % 224
x2 = x2 + 32
[x2] = v10        % 256
x2 = x2 + 32
[x2] = v11        % 288
x2 = x2 + 32
[x2] = v12        % 320
% vector operations do not change the flags:
x5 = x1 + 1       % 1: Z = 0
v13 = v1 - v2     % every lane is 0, but Z stays 0
BEQ .skip         % so this branch is not taken
x6 = 1
.skip
x2 = x2 + 32
[x2] = x6         % 352: 1 if the flags were left alone
