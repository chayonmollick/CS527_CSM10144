% Lab 5: touch data pages that are not in the data file. The loader maps only the
% pages the data file uses; any other page gets a free frame on first access.
% Writes 7 into pages 1, 3 and 7 of the data space, then stores their sum at 0.
x1 = 7
x2 = 255
x3 = x2 * 4       % address 1020 (data page 1)
[x3] = x1
x4 = x2 * 8       % address 2040 (data page 3)
[x4] = x1
x5 = x2 * 15      % address 3825 (data page 7)
[x5] = x1
x6 = [x3]
x7 = [x4]
x8 = [x5]
x9 = x6 + x7
x9 = x9 + x8
[0] = x9
