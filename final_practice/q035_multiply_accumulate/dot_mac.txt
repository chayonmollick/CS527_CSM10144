% Q035 test: dot product with multiply-accumulate (one instruction per element instead of two)
% data: N at 0, A at 4, B at 4 + 4N; the result goes to 4 + 8N
x1 = [0]          % N
x2 = 4            % address of A[i]
x3 = x1 * 4
x3 = x3 + 4       % address of B[i]
x4 = 0            % sum
.loop
x5 = [x2]
x6 = [x3]
x4 += x5 * x6     % sum = sum + A[i] * B[i]
x2 = x2 + 4
x3 = x3 + 4
x1 = x1 - 1
BNE .loop
[x3] = x4         % x3 = 4 + 8N = 36 now: 1*5 + (-2)*6 + 3*7 + 4*8 = 46
x7 = 3
x9 = 5
x9 = x9 - 5       % Z = 1
x7 += x7 * x7     % the destination is also a source: 3 + 3*3 = 12 (and Z must stay 1)
BEQ .kept
x7 = 0            % runs only if MAC changed the flags
.kept
[40] = x7         % 12
