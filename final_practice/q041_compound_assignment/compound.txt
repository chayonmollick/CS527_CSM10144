% Q041 test: compound assignment is compiler sugar: "x1 op= y" means "x1 = x1 op y"
x1 = 10
x2 = 3
x3 = 4
x1 += x2          % 13
x1 -= 5           % 8
x1 *= x3          % 32
x1 /= 2           % 16
[0] = x1
x4 = 0            % sum of 10..1 with compound statements
x5 = 10
.loop
x4 += x5
x5 -= 1           % a SUBI, so it sets Z for BNE
BNE .loop
[4] = x4          % 55
v1 = [32]         % vectors too: 1..8
v1 *= 3
v1 += v1          % 6 times the input
[64] = v1         % 6, 12, ..., 48
