% Q026 test: xD = abs xA (sum of the absolute values of an array)
x1 = [0]           % n = 5
x2 = 4             % address of a[0]
x3 = 0             % sum
.loop
x4 = [x2]
x5 = abs x4
x3 = x3 + x5
x2 = x2 + 4
x1 = x1 - 1
BNE .loop
[24] = x3          % |3| + |-4| + |0| + |-10| + |6| = 23
x6 = [8]           % -4
x6 = abs x6        % in place
[28] = x6          % 4
x7 = [32]          % -2147483648 has no positive partner in 32 bits
x7 = abs x7
[36] = x7          % stays -2147483648 (it wraps)
