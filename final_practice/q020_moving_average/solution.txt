% Q020 - moving average with a window of 3
% Task:   For an array of N integers (N >= 3) compute M[i] = (A[i] + A[i+1] + A[i+2]) / 3
%         for i = 0 .. N-3 (integer division, as the machine does it) into an
%         output array.
% Data:   N at 0 (N >= 3); A[0..N-1] at 4 .. 4N
% Result: M[0..N-3] (N - 2 values) at 4N + 4 .. 8N - 8
%         moving_average_data.byte : N = 13 -> 6 9 6 2 0 3 5 13 9 12 4 at 56 .. 96 (17 / 3 = 5)
%         moving_average_data2.byte: N = 5, negative sums -> -2 0 -2 at 24 .. 32 (-8 / 3 = -2)
% Method: VECTOR for the sums, scalar for the division. Like the Lab 3 FIR filter,
%         three loads shifted by one word, v1 = [&A[i]], v2 = [&A[i+1]],
%         v3 = [&A[i+2]], give 8 window sums with 2 vector adds; they are stored
%         in M. Sums left over (fewer than 8) are made one at a time. There is no
%         vector division, so a last scalar pass does M[i] = M[i] / 3, which
%         rounds towards zero (-8 / 3 = -2, not -3).
% Regs:   x1 N, x20 K = N - 2 outputs, x6 &M[0] = 4N + 4, x2 &A[i], x3 &M[i], x5 i,
%         x7 K - 8, x8 x10 &A[i+1] &A[i+2], x11 x12 values, x21 end of M,
%         v1 v2 v3 blocks
x1 = [0]          % N (>= 3)
x20 = x1 - 2      % K = N - 2 outputs
x6 = x1 * 4
x6 = x6 + 4       % &M[0] = 4N + 4, right after A
x2 = 4            % &A[i]
x3 = x6           % &M[i]
x5 = 0            % i
x7 = x20 - 8      % a full block of 8 outputs starts at i <= K - 8
.vloop
x9 = x5 - x7
BGT .tail         % fewer than 8 outputs left
x8 = x2 + 4
x10 = x2 + 8
v1 = [x2]         % A[i .. i+7]
v2 = [x8]         % A[i+1 .. i+8]
v3 = [x10]        % A[i+2 .. i+9]: still inside A, as i + 9 <= N - 1
v1 = v1 + v2
v1 = v1 + v3      % 8 window sums
[x3] = v1         % M[i .. i+7] = the sums (divided below)
x2 = x2 + 32
x3 = x3 + 32
x5 = x5 + 8
BAL .vloop
.tail
x9 = x5 - x20
BGE .divide       % all K sums are in M
x11 = [x2]        % A[i]
x8 = x2 + 4
x12 = [x8]        % A[i+1]
x11 = x11 + x12
x8 = x2 + 8
x12 = [x8]        % A[i+2]
x11 = x11 + x12
[x3] = x11        % M[i] = the sum (divided below)
x2 = x2 + 4
x3 = x3 + 4
x5 = x5 + 1
BAL .tail
.divide
x21 = x3          % &M[K]: the end of M
x3 = x6           % back to &M[0]
.dloop
x9 = x3 - x21
BGE .done
x11 = [x3]
x11 = x11 / 3     % scalar: the machine has no vector division
[x3] = x11
x3 = x3 + 4
BAL .dloop
.done
