% Q030 test: BZ xR .label and BNZ xR .label test a register; the flags are not used
x1 = [0]           % n = 5
x2 = 4             % address of a[0]
x3 = 0             % how many a[i] are 0
.next
x4 = [x2]
x2 = x2 + 4        % the flags now describe the address, not a[i]
BNZ x4 .nonzero
x3 = x3 + 1
.nonzero
x1 = x1 - 1
BNZ x1 .next
[24] = x3          % 2 zeros in 7 0 3 0 9
x5 = 0
x6 = x5 + 1        % Z = 0 now, but BZ looks at x5 itself
BZ x5 .skip
[28] = x6          % skipped
.skip
[32] = x6          % 1
