% Q013 - sum of the first N odd numbers, compared with N*N
% Task:   With a loop, compute S = 1 + 3 + 5 + ... + (2N-1), the sum of the first N
%         odd numbers. Also compute N*N, and store 1 if S == N*N, else 0.
% Data:   N at 0 (N >= 0)
% Result: S at 4, N*N at 8, the flag (S == N*N) at 12
%         odd_sum_data.byte : N = 300 -> [4] = 90000, [8] = 90000, [12] = 1
%         odd_sum_data2.byte: N = 0 (empty sum) -> [4] = 0, [8] = 0, [12] = 1
% Method: count-down loop run N times. The odd number grows by 2 per step, so no
%         multiply is needed inside the loop. "x3 = x3 - 1" sets Z when the
%         counter reaches 0, so BNE needs no separate compare. The guard before
%         the loop skips it for N = 0 (otherwise the count would go 0, -1, -2, ...).
% Regs:   x1 N, x2 current odd number, x3 counter, x10 S, x11 N*N, x12 flag
x1 = [0]          % N
x10 = 0           % S
x2 = 1            % first odd number
x3 = x1           % N steps to go
x9 = x3 - 0
BLE .done         % N <= 0: empty sum
.loop
x10 = x10 + x2    % S = S + odd
x2 = x2 + 2       % next odd number
x3 = x3 - 1
BNE .loop         % until the counter reaches 0
.done
[4] = x10         % S
x11 = x1 * x1
[8] = x11         % N*N
x12 = 0
x9 = x10 - x11
BNE .store        % S != N*N: flag stays 0
x12 = 1
.store
[12] = x12        % flag
