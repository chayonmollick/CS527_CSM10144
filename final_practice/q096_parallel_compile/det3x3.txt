% Lab 1 example program: determinant of a 3x3 matrix
% data: a b c d e f g h i (row-major) at 0..32; result at 36
% det = a(ei - fh) - b(di - fg) + c(dh - eg)
x1 = [0]          % a
x2 = [4]          % b
x3 = [8]          % c
x4 = [12]         % d
x5 = [16]         % e
x6 = [20]         % f
x7 = [24]         % g
x8 = [28]         % h
x9 = [32]         % i
x10 = x5 * x9
x11 = x6 * x8
x12 = x10 - x11   % ei - fh
x12 = x1 * x12
x10 = x4 * x9
x11 = x6 * x7
x13 = x10 - x11   % di - fg
x13 = x2 * x13
x10 = x4 * x8
x11 = x5 * x7
x14 = x10 - x11   % dh - eg
x14 = x3 * x14
x15 = x12 - x13
x15 = x15 + x14
[36] = x15
