% Q019 - element-wise product of two arrays, with vector instructions
% Task:   Compute C[i] = A[i] * B[i] for two arrays of N integers using vector
%         instructions. N is any number >= 0, not only a multiple of 8.
% Data:   N at 0; A[0..N-1] at 4 .. 4N; B[0..N-1] at 4N + 4 .. 8N
% Result: C[0..N-1] at 8N + 4 .. 12N
%         vec_product_data.byte : N = 19 (2 blocks + 3 left over), A = 1..19,
%                                 B[i] = 3i - 20 -> C = -20 -34 -42 ... 646 at 156 .. 228
%         vec_product_data2.byte: N = 5 (no full block) -> -120000 63 0 -144 3000 at 44 .. 60
% Method: VECTOR. While at least 8 elements are left (i <= N - 8), one load of A,
%         one load of B, one vector multiply and one store handle 8 elements. The
%         N mod 8 elements left over go through a scalar loop. Running one more
%         full block instead would write up to 7 words past the end of C.
% Regs:   x1 N, x2 &A[i], x3 &B[i], x4 &C[i], x5 i, x7 N - 8, x10 x11 one element,
%         v1 v2 blocks of 8
x1 = [0]          % N
x2 = 4            % &A[0]
x3 = x1 * 4
x3 = x3 + 4       % &B[0] = 4N + 4
x4 = x1 * 8
x4 = x4 + 4       % &C[0] = 8N + 4
x5 = 0            % i
x7 = x1 - 8       % a full block of 8 starts at i <= N - 8
.vloop
x9 = x5 - x7
BGT .tail         % fewer than 8 left
v1 = [x2]         % A[i .. i+7]
v2 = [x3]         % B[i .. i+7]
v1 = v1 * v2      % 8 products at once
[x4] = v1         % C[i .. i+7]
x2 = x2 + 32      % 8 words = 32 bytes
x3 = x3 + 32
x4 = x4 + 32
x5 = x5 + 8
BAL .vloop
.tail
x9 = x5 - x1
BGE .done         % i >= N
x10 = [x2]        % one element at a time
x11 = [x3]
x10 = x10 * x11
[x4] = x10
x2 = x2 + 4
x3 = x3 + 4
x4 = x4 + 4
x5 = x5 + 1
BAL .tail
.done
