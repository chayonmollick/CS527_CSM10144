% mat_sub_vec.txt - C = A - B and D = 2A - 3B for 4x4 matrices, with vector instructions
% Task:   Subtract two 4x4 integer matrices, C = A - B, and compute the linear
%         combination D = 2A - 3B, using vector instructions.
% Data:   A (row-major, 16 words) at 0 .. 60; B at 64 .. 124
% Result: C = A - B at 128 .. 188; D = 2A - 3B at 192 .. 252
%         mat_sub_vec_data.byte -> C = 4 0 -6 10 / -6 5 0 -4 / 12 0 -3 2 / 2 -4 0 3
%                                  D = 7 -3 -16 22 / -18 8 -1 -8 / 27 6 -11 3 / 2 -16 1 -1
% Method: VECTOR, no loop. A 4x4 matrix is 16 consecutive words = exactly 2 vectors
%         (rows 0-1 and rows 2-3), so element-wise work on a whole 4x4 matrix is
%         2 loads, 2 operations and 2 stores. All addresses are below 256, so constant
%         addresses ([0], [32], ...) work. "vD = vA * c" multiplies all 8 lanes by c.
% Variants: A + B or the element-wise product -> change "-" to "+" or "*";
%         8x8 -> one row = one vector, loop over the rows (programs/matrix_add_vec.txt);
%         any R x C -> blocks of 8 plus a scalar tail (scalar_mul_vec.txt);
%         a negative coefficient -> subtract the positive multiple (there is no "* -3").
% Regs:   v1 v2 A, v3 v4 B, v5 v6 C, v7 v8 2A, v9 v10 3B, v11 v12 D
v1 = [0]          % A rows 0-1 (A[0][0] .. A[1][3])
v2 = [32]         % A rows 2-3
v3 = [64]         % B rows 0-1
v4 = [96]         % B rows 2-3
v5 = v1 - v3      % C = A - B, rows 0-1
v6 = v2 - v4      % rows 2-3
[128] = v5
[160] = v6
v7 = v1 * 2       % 2A
v8 = v2 * 2
v9 = v3 * 3       % 3B
v10 = v4 * 3
v11 = v7 - v9     % D = 2A - 3B
v12 = v8 - v10
[192] = v11
[224] = v12
