% row_sums.txt - row sums of an R x C matrix, and the row with the largest sum
% Task:   For an R x C integer matrix A (R >= 1) compute every row sum
%         S[i] = A[i][0] + ... + A[i][C-1], then the largest row sum and its row index
%         (the first such row if several rows tie).
% Data:   R at 0, C at 4, A (row-major, R*C words) at 8 .. 4 + 4RC
% Result: S[0..R-1] right after A, at 8 + 4RC; then the largest sum at 8 + 4RC + 4R
%         and its row at 12 + 4RC + 4R
%         row_sums_data.byte : R = 3, C = 4 -> S at 56 .. 64 = 7 18 11; [68] = 18, [72] = 1
%         row_sums_data2.byte: R = 4, C = 2, all sums negative, a tie
%                              -> S at 40 .. 52 = -3 -3 -2 -3; [56] = -2, [60] = 2
% Method: SCALAR nested loops. A is read in order by one pointer (x3 + 4), so no
%         address arithmetic is needed. After each row: store S[i], then keep it as the
%         best if it is the first row or larger than the best so far (BLE skips equal
%         sums, which keeps the first row of a tie). Row sums add across one row, which
%         vectors cannot do without a lane sum (col_sums_vec.txt is the vector case).
% Variants: smallest row -> BGE instead of BLE; row averages -> x10 / C;
%         trace -> add only A[i][i] (pointer step 4(N+1), see final_practice/q015).
% Regs:   x1 R, x15 C, x2 i, x5 j, x3 &A[i][j], x4 &S[i], x6 element, x10 row sum,
%         x12 largest sum, x13 its row
x1 = [0]          % R
x15 = [4]         % C
x3 = 8            % &A[0][0]: walks through A in order
x4 = x1 * x15
x4 = x4 * 4
x4 = x4 + 8       % &S[0] = 8 + 4RC, right after A
x12 = 0           % largest row sum (set by row 0)
x13 = 0           % its row
x2 = 0            % i
.rows
x9 = x2 - x1
BGE .done         % i >= R
x10 = 0           % sum of row i
x5 = 0            % j
.cols
x9 = x5 - x15
BGE .rowdone      % j >= C
x6 = [x3]         % A[i][j]
x10 = x10 + x6
x3 = x3 + 4       % next element of A
x5 = x5 + 1
BAL .cols
.rowdone
[x4] = x10        % S[i]
x9 = x2 - 0
BEQ .take         % row 0: always the best so far
x9 = x10 - x12
BLE .next         % not larger: keep the earlier row
.take
x12 = x10
x13 = x2
.next
x4 = x4 + 4
x2 = x2 + 1
BAL .rows
.done
[x4] = x12        % x4 = &S[R]: the largest row sum
x4 = x4 + 4
[x4] = x13        % its row index
