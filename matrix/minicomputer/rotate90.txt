% rotate90.txt - rotate an N x N matrix 90 degrees clockwise in place (4x4 in the data)
% Task:   Rotate a 4x4 integer matrix by 90 degrees clockwise, in place: the first row
%         becomes the last column, i.e. the new A[j][N-1-i] is the old A[i][j].
% Data:   N at 0 (4 for a 4x4 matrix), A (row-major, N*N words) at 4 .. 4N^2
% Result: the rotated matrix in the same place, 4 .. 4N^2
%         rotate90_data.byte : N = 4, A = 1..16 -> 13 9 5 1 / 14 10 6 2 / 15 11 7 3 / 16 12 8 4
%         rotate90_data2.byte: N = 3, A = 1..9  -> 7 4 1 / 8 5 2 / 9 6 3
% Method: SCALAR, two passes, both in place (MATRIX_RULES.md section 6):
%         1. transpose: swap A[i][j] and A[j][i] for j > i (as transpose_inplace.txt);
%         2. reverse every row: swap the left and right ends and move both pointers
%            inwards until they meet (the middle element of an odd row stays).
%         Check on [[1,2],[3,4]]: transpose [[1,3],[2,4]], reverse rows [[3,1],[4,2]].
% Variants: anticlockwise -> transpose, then reverse every COLUMN (swap rows i and
%         N-1-i), or reverse the rows first and then transpose; 180 degrees -> reverse
%         the whole N*N array (programs/reverse_array.txt); out of place -> B[j][N-1-i] =
%         A[i][j]: copy programs/transpose.txt and start column i of B at N-1-i.
% Regs:   x1 N, x15 row length 4N, x2 i, x5 j, x3 x4 the two pointers, x6 x7 values
x1 = [0]          % N
x15 = x1 * 4      % one row = 4N bytes
x2 = 0            % pass 1: transpose in place
.trows
x9 = x2 - x1
BGE .reverse      % i >= N: transpose done
x3 = x2 * x15
x4 = x2 * 4
x3 = x3 + x4
x3 = x3 + 4       % &A[i][i]
x4 = x3
x5 = x2 + 1       % j = i + 1
.tcols
x9 = x5 - x1
BGE .tnext        % j >= N
x3 = x3 + 4       % &A[i][j]
x4 = x4 + x15     % &A[j][i]
x6 = [x3]
x7 = [x4]
[x3] = x7
[x4] = x6
x5 = x5 + 1
BAL .tcols
.tnext
x2 = x2 + 1
BAL .trows
.reverse
x2 = 0            % pass 2: reverse every row
.rrows
x9 = x2 - x1
BGE .done         % i >= N
x3 = x2 * x15
x3 = x3 + 4       % left = &A[i][0]
x4 = x3 + x15
x4 = x4 - 4       % right = &A[i][N-1]
.rswap
x9 = x3 - x4
BGE .rnext        % the pointers met or crossed
x6 = [x3]
x7 = [x4]
[x3] = x7
[x4] = x6
x3 = x3 + 4
x4 = x4 - 4
BAL .rswap
.rnext
x2 = x2 + 1
BAL .rrows
.done
