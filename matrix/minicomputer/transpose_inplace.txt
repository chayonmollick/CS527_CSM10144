% transpose_inplace.txt - transpose an N x N matrix in place (4x4 in the data)
% Task:   Replace a 4x4 integer matrix A by its transpose without a second matrix
%         (A[i][j] and A[j][i] change places).
% Data:   N at 0 (4 for a 4x4 matrix), A (row-major, N*N words) at 4 .. 4N^2
% Result: A^T in the same place, 4 .. 4N^2
%         transpose_inplace_data.byte : N = 4, A = 1..16 -> 1 5 9 13 / 2 6 10 14 / 3 7 11 15 / 4 8 12 16
%         transpose_inplace_data2.byte: N = 3, A = 1 -2 3 / 4 5 -6 / 7 8 9 -> 1 4 7 / -2 5 8 / 3 -6 9
% Method: SCALAR nested loops over the pairs ABOVE the diagonal only (j > i) and swap
%         each element with its mirror image. Swapping every pair (all j) would swap
%         each pair twice and give back the original matrix. For row i, p starts at
%         &A[i][i] and steps right (+4) to &A[i][j], q starts at &A[i][i] and steps
%         down (+4N) to &A[j][i]. The diagonal never moves.
% Variants: out of place -> programs/transpose.txt; symmetric check -> compare instead of
%         swap (final_practice/q018); rotate 90 degrees -> transpose, then reverse each
%         row (rotate90.txt).
% Regs:   x1 N, x15 row length 4N, x2 i, x5 j, x3 p = &A[i][j], x4 q = &A[j][i],
%         x6 x7 the two values
x1 = [0]          % N
x15 = x1 * 4      % one row = 4N bytes
x2 = 0            % i
.rows
x9 = x2 - x1
BGE .done         % i >= N
x3 = x2 * x15
x4 = x2 * 4
x3 = x3 + x4
x3 = x3 + 4       % &A[i][i] = 4 + 4Ni + 4i
x4 = x3           % q starts on the diagonal too
x5 = x2 + 1       % j = i + 1
.cols
x9 = x5 - x1
BGE .nextrow      % j >= N
x3 = x3 + 4       % p = &A[i][j]: one column right
x4 = x4 + x15     % q = &A[j][i]: one row down
x6 = [x3]
x7 = [x4]
[x3] = x7         % swap A[i][j] and A[j][i]
[x4] = x6
x5 = x5 + 1
BAL .cols
.nextrow
x2 = x2 + 1
BAL .rows
.done
