% is_upper.txt - is an N x N matrix upper triangular?
% Task:   Store 1 if the N x N integer matrix A is upper triangular (A[i][j] = 0 for
%         every i > j, i.e. everything below the main diagonal is 0), otherwise 0.
%         Also store how many elements below the diagonal are not 0.
% Data:   N at 0, A (row-major, N*N words) at 4 .. 4N^2
% Result: 1 or 0 at 4 + 4N^2, the count at 8 + 4N^2
%         is_upper_data.byte : upper triangular 4x4               -> [68] = 1, [72] = 0
%         is_upper_data2.byte: A[2][0] = 3 and A[3][1] = -1 below -> [68] = 0, [72] = 2
% Method: SCALAR nested loops over the strict lower triangle only: for i = 1..N-1,
%         j = 0..i-1. Row i starts at 4 + 4Ni and its part below the diagonal has i
%         elements, so the inner loop ends at j = i (half the matrix is never read).
%         Everything is counted (no early exit), and the answer is 1 when the count is 0.
% Variants: lower triangular -> for j = i+1..N-1 (start at &A[i][i+1]); strictly upper
%         (the diagonal must be 0 too) -> i from 0 and j up to i (x9 = x5 - x2, BGT);
%         diagonal matrix -> both triangles; early exit -> BNE .no at the first nonzero.
% Regs:   x1 N, x15 row length 4N, x2 i, x5 j, x3 &A[i][j], x6 element,
%         x11 nonzeros below the diagonal, x10 answer, x4 result address
x1 = [0]          % N
x15 = x1 * 4      % one row = 4N bytes
x11 = 0           % nonzeros below the diagonal
x2 = 1            % i: row 0 has nothing below the diagonal
.rows
x9 = x2 - x1
BGE .done         % i >= N
x3 = x2 * x15
x3 = x3 + 4       % &A[i][0] = 4 + 4Ni
x5 = 0            % j
.cols
x9 = x5 - x2
BGE .nextrow      % j >= i: the rest of row i is on or above the diagonal
x6 = [x3]         % A[i][j], i > j
x9 = x6 - 0
BEQ .zero
x11 = x11 + 1     % a nonzero below the diagonal
.zero
x3 = x3 + 4
x5 = x5 + 1
BAL .cols
.nextrow
x2 = x2 + 1
BAL .rows
.done
x10 = 0
x9 = x11 - 0
BNE .store        % something below the diagonal is not 0
x10 = 1
.store
x4 = x1 * x15
x4 = x4 + 4       % 4 + 4N^2, right after A
[x4] = x10
x4 = x4 + 4
[x4] = x11
