% count_zeros.txt - zero and nonzero elements of an R x C matrix, and all-zero rows
% Task:   For an R x C integer matrix count the elements that are 0, the elements that
%         are not 0, and the rows whose elements are all 0.
% Data:   R at 0, C at 4, A (row-major, R*C words) at 8 .. 4 + 4RC
% Result: zeros at 8 + 4RC, nonzeros at 12 + 4RC, all-zero rows at 16 + 4RC
%         count_zeros_data.byte : R = 4, C = 5, rows 1 and 3 all zero -> [88] = 15, [92] = 5, [96] = 2
%         count_zeros_data2.byte: R = 3, C = 3, no zero at all       -> [44] = 0, [48] = 9, [52] = 0
% Method: SCALAR nested loops, A read in order by one pointer. "x9 = x6 - 0" sets the
%         flags from the element (a load does not), then BNE skips the count. A row
%         is all zero when its zero count equals C. Nonzeros = R*C - zeros.
%         (A matrix with mostly zeros is "sparse": store only the nonzeros, as CSR.)
% Variants: count negatives / positives -> BMI / BGT after "x9 = x6 - 0"; count a key k
%         -> "x9 = x6 - x20" with k in x20; all-zero columns -> walk down each column (+4C).
% Regs:   x1 R, x15 C, x2 i, x5 j, x3 &A[i][j], x6 element, x10 zeros, x11 nonzeros,
%         x12 zero rows, x13 zeros in row i, x16 R*C
x1 = [0]          % R
x15 = [4]         % C
x3 = 8            % &A[0][0]: walks through A in order
x10 = 0           % zeros
x12 = 0           % all-zero rows
x2 = 0            % i
.rows
x9 = x2 - x1
BGE .done         % i >= R
x13 = 0           % zeros in row i
x5 = 0            % j
.cols
x9 = x5 - x15
BGE .rowdone      % j >= C
x6 = [x3]         % A[i][j]
x9 = x6 - 0       % flags from the element
BNE .nonzero
x13 = x13 + 1
.nonzero
x3 = x3 + 4
x5 = x5 + 1
BAL .cols
.rowdone
x10 = x10 + x13
x9 = x13 - x15
BNE .nextrow      % fewer than C zeros: not an all-zero row
x12 = x12 + 1
.nextrow
x2 = x2 + 1
BAL .rows
.done
x16 = x1 * x15    % R*C
x11 = x16 - x10   % nonzeros
[x3] = x10        % x3 = 8 + 4RC now: right after A
x3 = x3 + 4
[x3] = x11
x3 = x3 + 4
[x3] = x12
