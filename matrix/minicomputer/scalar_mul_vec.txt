% scalar_mul_vec.txt - B = k*A for an R x C matrix of any size, with vector instructions
% Task:   Multiply every element of an R x C integer matrix A by a scalar k (B = kA).
% Data:   R at 0, C at 4, k at 8, A (row-major, R*C words) at 12 .. 8 + 4RC
% Result: B (row-major) right after A, at 12 + 4RC
%         scalar_mul_vec_data.byte : R = 3, C = 5, k = 3 (15 words: 1 block + 7 left over)
%                                    -> B at 72 .. 128 = 6 -3 0 12 21 / 9 15 -6 24 3 / -18 27 30 0 -9
%         scalar_mul_vec_data2.byte: R = 4, C = 4, k = -2, A = 1..16 (2 blocks, nothing left)
%                                    -> B at 76 .. 136 = -2 -4 -6 .. -32
% Method: VECTOR. A matrix stored row after row is one array of N = R*C words, so the
%         rows do not matter for element-wise work. Blocks of 8 use "v1 = v1 * x5" (k in a
%         scalar register, any value, also negative) while i <= N - 8; a scalar loop does
%         the last N mod 8 elements.
% Variants: kA + B -> load the B block too and add it; in place (A = kA) -> store at x3;
%         a constant k (0..255) -> "v1 = v1 * 3" needs no register.
% Regs:   x10 R, x11 C, x5 k, x1 N = R*C, x7 N - 8, x2 i, x3 &A[i], x4 &B[i], x6 one element,
%         v1 one block of 8
x10 = [0]         % R
x11 = [4]         % C
x5 = [8]          % k
x1 = x10 * x11    % N = R*C elements
x3 = 12           % &A[0]
x4 = x1 * 4
x4 = x4 + 12      % &B[0] = 12 + 4N, right after A
x7 = x1 - 8       % a full block of 8 starts at i <= N - 8
x2 = 0            % i
.vloop
x9 = x2 - x7
BGT .tail         % i > N - 8: fewer than 8 elements left
v1 = [x3]         % A[i .. i+7]
v1 = v1 * x5      % 8 products k*A at once
[x4] = v1         % B[i .. i+7]
x3 = x3 + 32      % next block: 8 words = 32 bytes
x4 = x4 + 32
x2 = x2 + 8
BAL .vloop
.tail
x9 = x2 - x1
BGE .done         % i >= N
x6 = [x3]
x6 = x6 * x5
[x4] = x6
x3 = x3 + 4
x4 = x4 + 4
x2 = x2 + 1
BAL .tail
.done
